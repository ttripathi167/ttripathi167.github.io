from fastapi import FastAPI, Depends
from pydantic import BaseModel

# --- MCP tool: grounded retrieval the agent can call ---
@mcp.tool()
async def retrieve_incident_context(query: str, top_k: int = 8) -> list[Chunk]:
    """Hybrid dense+BM25 search, reranked, with citations."""
    hits = await vector_db.hybrid_search(query, k=top_k * 3)
    ranked = reranker.rerank(query, hits)[:top_k]
    return [assemble_with_citations(c) for c in ranked]

class TriageRequest(BaseModel):
    incident_id: str
    priority: str = "P2"

@app.post("/v1/agents/triage")
async def triage(req: TriageRequest, user=Depends(require_oauth2)):
    plan = await orchestrator.plan(req, tools=[retrieve_incident_context])
    if plan.risk == "high":
        await hitl_gate.request_approval(plan, approver=user.team)  # human-in-the-loop
    result = await orchestrator.execute(plan, model="claude")
    await telemetry.log_trace(result, eval_suite="groundedness-v3")
    return {"status": "ok", "citations": result.citations}
